cmake_minimum_required(VERSION 3.22)
project(hoonbot C CXX)

# RakNet 2.52 era code: needs C++14 (uses 'register')
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

file(GLOB RAKNET_SRC ${CMAKE_SOURCE_DIR}/raknet/*.cpp)
# server/email/telnet/etc. are not needed for a client and do not build on clang
list(FILTER RAKNET_SRC EXCLUDE REGEX ".*/(EmailSender|FullyConnectedMesh|NetworkIDGenerator|ConsoleServer|TelnetTransport|TCPInterface|RakNetCommandParser|LogCommandParser|CommandParserInterface|AsynchronousFileIO|ExtendedOverlappedPool|SystemDatabaseClient|SystemDatabaseServer|TableSerializer|DS_Table|_findfirst|FileOperations)\\.cpp$")

# AuthKeyTable lives in raknet/SAMP/samp_auth.cpp
file(GLOB SAMP_SRC ${CMAKE_SOURCE_DIR}/raknet/SAMP/*.cpp)

add_library(hoonbot SHARED hoonbot.cpp ${RAKNET_SRC} ${SAMP_SRC})
target_include_directories(hoonbot PRIVATE ${CMAKE_SOURCE_DIR}/raknet ${CMAKE_SOURCE_DIR}/raknet/SAMP)
target_compile_definitions(hoonbot PRIVATE RAKSAMP_CLIENT)
target_compile_options(hoonbot PRIVATE -w -fvisibility=hidden
    $<$<COMPILE_LANGUAGE:CXX>:-include${CMAKE_SOURCE_DIR}/raknet/hoon_compat.h>)
target_link_options(hoonbot PRIVATE -Wl,--exclude-libs,ALL)
target_link_libraries(hoonbot log)
